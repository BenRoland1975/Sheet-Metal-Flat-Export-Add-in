    Private Sub LoadShippingLabels()
        Try
            UpdateStatus("Loading shipping labels...")
            ShowProgress(True)
            
            ' Clear existing data
            shippingData.Clear()
            
            If swModel Is Nothing OrElse swModel.GetType() <> swDocumentTypes_e.swDocDRAWING Then
                UpdateStatus("Please open a drawing document")
                ShowProgress(False)
                Return
            End If
            
            Dim swDraw As SldWorks.DrawingDoc = swModel
            Dim swSketchMgr As SldWorks.SketchManager = swModel.SketchManager
            
            ' Extract shipping labels using the existing extraction logic
            Dim shippingLabels() As ShippingLabelData
            Call ExtractAllShippingLabelsUI(swDraw, swSketchMgr, shippingLabels)
            
            ' Populate data table
            If UBound(shippingLabels) >= 0 Then
                For i As Integer = 0 To UBound(shippingLabels)
                    Dim row As DataRow = shippingData.NewRow()
                    With shippingLabels(i)
                        row("InstanceID") = .InstanceName
                        row("JobNumber") = .JobNumber
                        row("SystemNumber") = .SystemNumber
                        row("IDNumber") = .IDNumber
                        row("Description") = .Description
                        row("Quantity") = .Quantity
                        row("XPosition") = .XPosition
                        row("YPosition") = .YPosition
                        row("SheetName") = .SheetName
                        row("HasLeader") = .HasLeader
                        row("ComponentName") = .ComponentName
                        row("PartFileName") = .PartFileName
                        row("MaterialName") = .MaterialName
                        row("PartNumber") = .PartNumber
                        row("PartDescription") = .Description_Part
                        row("Mass") = .Mass
                    End With
                    shippingData.Rows.Add(row)
                Next
            End If
            
            UpdateStatus($"Loaded {shippingData.Rows.Count} shipping labels")
            ShowProgress(False)
            
        Catch ex As Exception
            MessageBox.Show($"Error loading shipping labels: {ex.Message}", "Error", MessageBoxButtons.OK, MessageBoxIcon.Error)
            UpdateStatus("Error loading data")
            ShowProgress(False)
        End Try
    End Sub
    
    Private Sub UpdateStatus(message As String)

    Private Sub dgvShippingLabels_CellEndEdit(sender As Object, e As DataGridViewCellEventArgs) Handles dgvShippingLabels.CellEndEdit
        ' Update the SOLIDWORKS block when user edits a cell
        Try
            Dim row As DataGridViewRow = dgvShippingLabels.Rows(e.RowIndex)
            Dim instanceID As String = row.Cells("InstanceID").Value.ToString()
            Dim columnName As String = dgvShippingLabels.Columns(e.ColumnIndex).DataPropertyName
            Dim newValue As String = row.Cells(e.ColumnIndex).Value?.ToString() ?? ""
            
            ' Update the block attribute in SOLIDWORKS
            UpdateShippingLabelAttributeUI(instanceID, columnName, newValue)
            
            UpdateStatus($"Updated {columnName} for {instanceID}")
            
        Catch ex As Exception
            MessageBox.Show($"Error updating block: {ex.Message}", "Update Error", MessageBoxButtons.OK, MessageBoxIcon.Error)
            LoadShippingLabels() ' Reload to revert changes
        End Try
    End Sub
    
    ' Utility Methods
    Private Sub ExportDataTableToCSV(dt As DataTable, filePath As String)
        Using writer As New System.IO.StreamWriter(filePath)
            ' Write headers
            Dim headers() As String = dt.Columns.Cast(Of DataColumn)().Select(Function(column) column.ColumnName).ToArray()
            writer.WriteLine(String.Join(",", headers))
            
            ' Write data rows
            For Each row As DataRow In dt.Rows
                Dim fields() As String = row.ItemArray.Select(Function(field) $"""{field}""").ToArray()
                writer.WriteLine(String.Join(",", fields))
            Next
        End Using
    End Sub

    Private Sub UpdateShippingLabelAttributeUI(instanceID As String, attributeName As String, newValue As String)
        ' Map UI column names to block attribute names
        Dim blockAttributeName As String = ""
        Select Case attributeName
            Case "JobNumber"
                blockAttributeName = "JOB_NUMBER"
            Case "SystemNumber"
                blockAttributeName = "SYSTEM_NUMBER"
            Case "IDNumber"
                blockAttributeName = "ID_NUMBER"
            Case "Description"
                blockAttributeName = "DESCRIPTION"
            Case "Quantity"
                blockAttributeName = "QUANTITY"
            Case Else
                Return ' Don't update read-only or unmapped fields
        End Select
        
        ' Call the existing update function
        UpdateShippingLabelAttribute(instanceID, blockAttributeName, newValue)
    End Sub
